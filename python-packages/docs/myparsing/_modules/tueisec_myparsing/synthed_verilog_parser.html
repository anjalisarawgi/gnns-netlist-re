class VerilogParser(PLYParser):
    'Verilog HDL Parser'

    # Expression Precedence
    # Reference: http://hp.vector.co.jp/authors/VA016670/verilog/index.html
    # precedence = (
    #     # <- Weak
    #     ('left', 'LOR'),
    #     ('left', 'LAND'),
    #     ('left', 'OR'),
    #     ('left', 'AND', 'XOR', 'XNOR'),
    #     ('left', 'EQ', 'NE', 'EQL', 'NEL'),
    #     ('left', 'LT', 'GT', 'LE', 'GE'),
    #     ('left', 'LSHIFT', 'RSHIFT', 'LSHIFTA', 'RSHIFTA'),
    #     ('left', 'PLUS', 'MINUS'),
    #     ('left', 'TIMES', 'DIVIDE', 'MOD'),
    #     ('left', 'POWER'),
    #     ('right', 'UMINUS', 'UPLUS', 'ULNOT', 'UNOT',
    #      'UAND', 'UNAND', 'UOR', 'UNOR', 'UXOR', 'UXNOR'),
    #     # -> Strong
    # )

def parse(filepath, closefd=True, debug=False):
    """Parse a single synthesized verilog netlist file to AST-representation.

    :param filepath: Path to the file that should be parsed.
    :type filepath: str
    :param closefd: Forwarded to the :py:func:`open` call, opening the filepath.
    :type closefd: bool
    :param debug: Enable debug otuput of lexer and parser, defaults to False
    :type debug: bool, optional
    :return: Returns an AST representation of the parsed verilog file.
    :rtype: synthed_verilog_ast.Source
    """
    codeparser = VerilogFileParser(filepath)
    ast = codeparser.parse(closefd=closefd, debug=debug)
    directives = codeparser.get_directives()
    return ast, directives
