    def show(self, buf=sys.stdout, offset=0, attrnames=False, showlineno=True):
        indent = 2
        lead = ' ' * offset
        buf.write(lead + self.__class__.__name__ + ': ')
        if self.attr_names:
            if attrnames:
                nvlist = [(n, getattr(self, n)) for n in self.attr_names]
                attrstr = ', '.join('%s=%s' % (n, v) for (n, v) in nvlist)
            else:
                vlist = [getattr(self, n) for n in self.attr_names]
                attrstr = ', '.join('%s' % v for v in vlist)
            buf.write(attrstr)
        if showlineno:
            buf.write(' (at %s)' % self.lineno)
        buf.write('\n')
        for c in self.children():
            c.show(buf, offset + indent, attrnames, showlineno)



    def __eq__(self, other):
        if type(self) != type(other):
            return False
        self_attrs = tuple([getattr(self, a) for a in self.attr_names])
        other_attrs = tuple([getattr(other, a) for a in other.attr_names])
        if self_attrs != other_attrs:
            return False
        other_children = other.children()
        for i, c in enumerate(self.children()):
            if c != other_children[i]:
                return False
        return True

    def __ne__(self, other):
        return not self.__eq__(other)

    def __hash__(self):
        s = hash(tuple([getattr(self, a) for a in self.attr_names]))
        c = hash(self.children())
        return hash((s, c))

    def __str__(self):
        return ASTCodeGenerator().visit(self)

class IntConst(Constant):
    base_types = {
        "b": 2,
        "o": 8,
        "h": 16,
        "d": 10,
    }

    def __index__(self):
        try:
            bits, value = self.value.split("'")
            base_string = value[0]
            base = self.base_types.get(base_string.lower(), 10)
            value = value[1:].replace("_","")
        except ValueError as e:
            return int(self.value)
        return int(value, base)

    def __int__(self):
        return self.__index__()



[docs]
    def bits(self):
        try:
            bits, value = self.value.split("'")
        except ValueError:
            try:
                bits = int(self.value)
            except ValueError:
                if self.value in ["x", "X", "z", "Z"]:
                    return 1
                else:
                    raise ValueError(
                        "The value {} is not containing any bit information.".format(
                                self.value
                            )
                        )
            return 1 if bits < 1 else math.ceil(math.log2(bits))
        return int(bits)



    def __lt__(self, other):
        if isinstance(other, type(self)):
            return int(self) < int(other)
        elif isinstance(other, int):
            return int(self) < int(other)
        else:
            return NotImplemented

    def __gt__(self, other):
        if isinstance(other, type(self)):
            return int(self) > int(other)
        elif isinstance(other, int):
            return int(self) > int(other)
        else:
            return NotImplemented

    def __eq__(self, other):
        if isinstance(other, type(self)):
            return int(self) == int(other)
        elif isinstance(other, int):
            return int(self) == int(other)
        else:
            return NotImplemented

    def __add__(self, other):
        if isinstance(other, IntConst):
            return int(self) + int(other)
        elif isinstance(other, FloatConst):
            return float(self) + float(other.value)
        else:
            return NotImplemented

    def __radd__(self, other):
        if isinstance(other, IntConst):
            return int(other)+ int(self)
        elif isinstance(other, FloatConst):
            return float(other.value) + float(self)
        elif isinstance(other, Integral):
            return int(other) + int(self)
        elif isinstance(other, Real):
            return float(other) + float(self)
        elif isinstance(other, Complex):
            return complex(other) + complex(self)
        else:
            return NotImplemented
