def call_tmax(filename, reportpath, topname, libPath):
    """generate a scriptfile in tempdir and then call tmax with the right commands.

    :param filename: filepath of the verilog infile
    :type filename: str
    :param reportpath: Filename for the report outputfile.
    :type reportpath: str
    :param topname: Name of the topmodule, tmax needs this to parse the verilog
    :type topname: str
    :param libPath: verilog file that tmax reads as a cell library,
      defaults to LIB_PATH_DEFAULT
    :type libPath: str, optional
    :return: reportpath
    :rtype: str
    """

    glogger.info("Writing shell file")
    x = []
    # x.append("cd {}\n".format(os.path.dirname(outfilepath)))
    x.append("module load synopsys/tx/F-2011.09-SP3\n")
    x.append(
        "tmax -shell {} -env VPATH '{}' -env TOPNAME '{}' "
        "-env LIBPATH '{}' > {}\n".format(
            REPORT_PINS_SCRIPT, filename, topname, libPath, reportpath
        )
    )
    with tempfile.NamedTemporaryFile(mode="w+", suffix=".sh", delete=False,) as tmpfile:
        tmpfile.writelines(x)
    os.chmod(tmpfile.name, stat.S_IXUSR | stat.S_IRUSR)
    glogger.info("Calling tmax")
    subprocess.run(tmpfile.name, shell=True, check=True)
    os.remove(tmpfile.name)

def analyse_report(reportpath):
    """Analyze the reportfile generated by tmax

    :param reportpath: path of the report file
    :type reportpath: str or path that can be `open`-ed
    :return: scoap dataframe
    :rtype: pandas.DataFrame
    """
    with open(reportpath) as f:
        report = f.readlines()

    # Extrahieren der benötigten Informationen aus der report-Datei und
    # ergänzen um Informationen der mod_sig-Datei
    start = 0
    tmp_data = []
    rn = 0
    for r in report:
        if start == 1:
            if (r.find(" I ") != -1 and r.find("---") == -1 and r.find("()") == -1) or (
                rn == 1 and r.find(" I ") != -1
            ):
                c = r[r.find("(") + 1 : r.find(")")]
                c = c.split("-")
                cc0 = c[0]  # 1. Wert
                cc1 = c[1]  # 2. Wert
                co = c[2]  # 3. Wert
                if r.find("..."):
                    ursprung = r[r.find(")") + 3 : r.find("...")]
                else:
                    ursprung = r[r.find(")") + 3 : r.find("/n")]
                if cc0 != "*" and cc1 != "*" and cc0 != "M" and cc1 != "M":
                    cc = math.sqrt(int(cc0) ** 2 + int(cc1) ** 2)
                else:
                    if cc0 == "M":
                        cc0 = cc = 295000
                    if cc1 == "M":
                        cc1 = cc = 295000
                    if cc0 == "*":
                        cc0 = cc = 300000
                    if cc1 == "*":
                        cc1 = cc = 300000
                if co == "M":
                    co = 295000
                elif co == "*":
                    co = 300000  # math.inf
                else:
                    co = int(co)
                tmp_data.append((ursprung, cc0, cc1, cc, co))
                rn = 0
            elif r.find("PO (_PO)") != -1:
                rn = 1
            else:
                rn = 0
        elif r.find("report_primi") != -1:
            start = 1

    co_cc_pairs = pd.DataFrame.from_records(
        tmp_data, columns=["cell", "CC0", "CC1", "CC", "CO"], index="cell"
    )

    return co_cc_pairs
